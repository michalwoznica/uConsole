# ClockworkPi uConsole CM4 distro-neutral Raspberry Pi firmware boot policy.
#
# Image targets install this file as /boot/config.txt.

[all]
arm_64bit=1
disable_overscan=1
ignore_lcd=1
max_framebuffers=2

# Keep the uConsole backlight enable asserted from firmware handoff until the
# OCP8178 backlight driver takes ownership of GPIO9.
gpio=9=op,dh

# uConsole headphone detect: input with no internal pull. Physical validation
# proved GPIO10 stays low with the default bias but tracks jack insertion after
# applying the equivalent runtime setting: pinctrl set 10 ip pn.
gpio=10=ip,np

# CM4 USB2 host path used by uConsole peripherals.
otg_mode=1

# Raspberry Pi PWM audio routed to uConsole GPIO12/13.
dtparam=audio=on
dtoverlay=audremap,pins_12_13

[cm4]
# Project-owned uConsole hardware description.
dtoverlay=clockworkpi-uconsole-cm4

# Raspberry Pi 4/CM4 KMS display path required by the DSI panel.
dtoverlay=vc4-kms-v3d-pi4,cma-384

# Project uConsole kernel image.
kernel=kernel8.img

# Direct ext4 root boot; no firmware-loaded initramfs.
# GPIO14/15 remain the uConsole expansion UART; no serial console is enabled.
